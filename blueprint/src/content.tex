\chapter{Dimension five graphs with fifteen edges}

\section{Definitions}

\begin{definition}[Unit-distance representation]
  \label{def:representable}
  \lean{Erdos1007Dim5.Standalone.Mathlib.InlineErdos1007Dim5.UnitDistanceEmbeddable, SimpleGraph.UnitDistEmbeddable}
  \leanok
  A graph $G$ is \emph{representable in $\mathbb{R}^n$} when there is an injective
  $f : V(G) \to \mathbb{R}^n$ with $\lVert f(u) - f(v)\rVert = 1$ for every edge $uv$.
  Non-adjacent vertices are unconstrained. The statement's inlined definition and the library's
  agree by \texttt{Iff.rfl}.
\end{definition}

\begin{definition}[Dimension]
  \label{def:dimension}
  \lean{Erdos1007Dim5.Standalone.Mathlib.InlineErdos1007Dim5.HasDimension, SimpleGraph.HasUnitDistDim}
  \leanok
  \uses{def:representable}
  $\dim(G) = n$ when $n$ is least such that $G$ is representable in $\mathbb{R}^n$.
\end{definition}

\section{From the library}

\begin{lemma}[Monotonicity; Chaffee--Noble Lemma 4 and Corollary 5]
  \label{lem:monotone}
  \lean{SimpleGraph.UnitDistEmbeddable.mono, SimpleGraph.UnitDistEmbeddable.comap, SimpleGraph.UnitDistEmbeddable.of_le}
  \leanok
  \uses{def:representable, def:dimension}
  A representation in $\mathbb{R}^n$ gives one in $\mathbb{R}^m$ for $m \ge n$, and restricts along
  graph homomorphisms and to subgraphs. (Corollary 5, $\dim H \le \dim G$ for a spanning subgraph, is
  the library's \texttt{dim\_le\_of\_le\_compl}; this proof does not need it.)
\end{lemma}

\begin{lemma}[Re-attaching a vertex with a clique neighbourhood]
  \label{lem:reattach}
  \lean{EuclideanGeometry.infinite_sphere_inter_of_regular_simplex, SimpleGraph.UnitDistEmbeddable.extend}
  \leanok
  \uses{def:representable}
  If the neighbours of $u$ form a clique of size $k$ with $k + 1 \le d$, and $G - u$ is representable
  in $\mathbb{R}^d$, then so is $G$. The points at distance one from a unit regular simplex with
  $k + 1 \le d$ vertices form an infinite set, so $u$ avoids the finitely many placed points.
\end{lemma}

\begin{lemma}[$\dim K_n = n - 1$; Chaffee--Noble Lemma 1]
  \label{lem:dim-complete}
  \lean{EuclideanGeometry.card_le_of_equilateral, SimpleGraph.hasUnitDistDim_completeGraph}
  \leanok
  \uses{def:dimension}
  At most $d + 1$ points of $\mathbb{R}^d$ are pairwise at distance one, and scaled basis vectors
  place $K_n$ in $\mathbb{R}^{n-1}$.
\end{lemma}

\begin{lemma}[$\dim (K_n - e) = n - 2$; Chaffee--Noble Lemma 2]
  \label{lem:dim-complete-minus-edge}
  \lean{SimpleGraph.hasUnitDistDim_completeGraph_deleteEdge}
  \leanok
  \uses{lem:dim-complete, lem:monotone}
  For $n \ge 3$: two apexes over the remaining $n - 2$ vertices for the upper bound, and
  $K_{n-1} \le K_n - e$ for the lower.
\end{lemma}

\begin{lemma}[$K_{3,3}$ has no representation in $\mathbb{R}^3$; Chaffee--Noble Lemma 3]

\section{Theorem 8: $\dim K_{1,3,3} = 5$}

\begin{theorem}[Chaffee--Noble, Theorem 8]
  \label{thm:k133-dimension}
  \lean{SimpleGraph.not_unitDistEmbeddable_completeMultipartiteGraph_one_three_three, SimpleGraph.unitDistEmbeddable_completeMultipartiteGraph_one_three_three, SimpleGraph.hasUnitDistDim_K133}
  \leanok
  \uses{lem:monotone}
  $K_{1,3,3}$ has no representation in $\mathbb{R}^4$ and has one in $\mathbb{R}^5$. The lower
  bound is a self-contained orthogonal-complement count in place of the paper's circumcentre
  argument; the upper bound is an explicit placement, not the paper's inclusion in $K_7 - e$.
\end{theorem}

\section{Lemma 9: the cocktail-party graphs}

\begin{lemma}[Chaffee--Noble, Lemma 9]
  \label{lem:cocktail}
  \lean{SimpleGraph.hasUnitDistDim_completeMultipartiteGraph_one_two_two_two, SimpleGraph.hasUnitDistDim_completeMultipartiteGraph_two_two_two_two}
  \leanok
  \uses{lem:k33-lower, lem:monotone}
  $K_{1,2,2,2}$ and $K_{2,2,2,2}$ have dimension four: cross-polytope placements
  $\pm \tfrac{\sqrt 2}{2} e_i$, and a $K_{3,3}$ homomorphism for the lower bound. The $K_{3,3}$ is
  not induced, so the lower bound goes through a homomorphism, not an embedding.
\end{lemma}

\section{Theorem 10: at most fourteen edges is representable in $\mathbb{R}^4$}

\begin{lemma}[Two-regular graphs on seven vertices]
  \label{lem:two-regular-seven}
  \lean{SimpleGraph.exists_three_pairwise_disjoint_edges, SimpleGraph.exists_three_pairwise_disjoint_edges_compl}
  \leanok
  A two-regular graph on seven vertices has three pairwise disjoint edges, since its components
  are cycles of length at least three. So the complement of a $4$-regular graph on seven
  vertices has three pairwise disjoint edges.
\end{lemma}

\begin{theorem}[Chaffee--Noble, Theorem 10, lower bound]
  \label{thm:fewer-than-fifteen}
  \lean{SimpleGraph.unitDistEmbeddable_four_of_ncard_edgeSet_le}
  \leanok
  \uses{lem:reattach, lem:monotone, lem:dim-complete, lem:dim-complete-minus-edge, lem:cocktail, lem:two-regular-seven}
  Every finite graph with at most fourteen edges is representable in $\mathbb{R}^4$.
\end{theorem}

\begin{proof}
  \leanok
  By induction on the number of vertices.
  \begin{itemize}
    \item A vertex of degree at most three reduces. Delete it and join its neighbours pairwise:
      at most as many edges and fewer vertices, so the smaller graph is representable, and
      Lemma~\ref{lem:reattach} re-attaches the vertex to that supergraph.
    \item Otherwise the minimum degree is at least four, so $4|V| \le 28$ and $|V| \le 7$.
      \begin{itemize}
        \item $|V| \le 5$: the graph is a subgraph of $K_5$.
        \item $|V| = 6$: it misses an edge of $K_6$, so it is a subgraph of $K_6 - e$.
        \item $|V| = 7$: it is $4$-regular with fourteen edges. Its complement has three
          disjoint edges (Lemma~\ref{lem:two-regular-seven}), so it maps into $K_{1,2,2,2}$.
      \end{itemize}
  \end{itemize}
\end{proof}

\section{The targets}

\begin{theorem}[Attainment; upstream \texttt{dimension\_five\_extremal}]
  \label{thm:attainment}
  \lean{Erdos1007Dim5.Standalone.Mathlib.InlineErdos1007Dim5.DimensionFiveExtremal.proof}
  \leanok
  \uses{lem:dim-complete, thm:k133-dimension}
  $K_6$ and $K_{1,3,3}$ each have dimension five and fifteen edges. No uniqueness is claimed.
\end{theorem}

\begin{theorem}[Chaffee--Noble, Theorem 10; upstream \texttt{dimension\_five}]
  \label{thm:dimension-five}
  \lean{Erdos1007Dim5.Standalone.Mathlib.InlineErdos1007Dim5.DimensionFive.proof}
  \leanok
  \uses{thm:fewer-than-fifteen, thm:attainment}
  The least number of edges of a graph of dimension five is fifteen. $K_6$ attains fifteen, and
  a graph with at most fourteen edges is representable in $\mathbb{R}^4$, so it does not have
  dimension five.
\end{theorem}
